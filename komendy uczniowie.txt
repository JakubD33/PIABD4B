Średnia cena produktu w całym sklepie (AVG)sql
SELECT AVG(cena) AS srednia_cena FROM produkty;

najdroższy i najtańszy produkt (MIN, MAX)sql
SELECT MIN(cena) AS najtanszy, MAX(cena) AS najdrozszy FROM produkty;

Łączna wartość towaru w magazynie (SUM z operacją matematyczną)sql
SELECT SUM(cena * magazyn) AS wartosc_magazynu FROM produkty;
